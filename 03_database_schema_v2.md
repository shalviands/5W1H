# Antigravity — Database Schema v2.0
## Extension of Existing INCUBX Supabase Schema

---

## PRINCIPLE

This schema does NOT replace or replicate existing INCUBX tables.
It extends them using foreign key references.

```
EXISTING INCUBX TABLES (DO NOT MODIFY)
  users          → referenced via user_id
  teams          → referenced via team_id
  cohorts        → referenced via cohort_id
  projects       → referenced via project_id

NEW ANTIGRAVITY TABLES (ADD THESE)
  ag_sessions    → one discovery session per project/team
  ag_insights    → append-only raw insight store
  ag_clusters    → grouped patterns (generated by ClusterAgent)
  ag_mind_maps   → synthesised output per session
  ag_prompt_versions → prompt versioning registry
```

---

## TABLE 1: ag_sessions

One row per discovery session. Linked to project + team + cohort.

```sql
CREATE TABLE ag_sessions (
  id                UUID PRIMARY KEY DEFAULT gen_random_uuid(),

  -- Platform references (foreign keys to existing tables)
  project_id        UUID NOT NULL REFERENCES projects(id) ON DELETE CASCADE,
  team_id           UUID NOT NULL REFERENCES teams(id) ON DELETE CASCADE,
  cohort_id         UUID NOT NULL REFERENCES cohorts(id) ON DELETE SET NULL,
  created_by        UUID NOT NULL REFERENCES users(id),

  -- Session config
  problem_domain    TEXT NOT NULL,
  stakeholders      TEXT[] NOT NULL DEFAULT '{}',
  session_type      TEXT NOT NULL CHECK (session_type IN ('individual', 'team', 'mentor_assisted')),
  session_language  TEXT NOT NULL DEFAULT 'en' CHECK (session_language IN ('en', 'hi', 'mixed')),

  -- Progress tracking
  status            TEXT NOT NULL DEFAULT 'active'
                    CHECK (status IN ('created', 'active', 'paused', 'resumed', 'synthesis_ready', 'completed')),
  covered_dimensions TEXT[] DEFAULT '{}',
  insight_count     INT DEFAULT 0,
  last_active_at    TIMESTAMPTZ,

  -- Resumability
  last_summary      TEXT,              -- 3-line summary for resume context
  conversation_history JSONB DEFAULT '[]',  -- full message history for context window

  -- Prompt version used
  prompt_version    TEXT NOT NULL DEFAULT 'v2',

  -- Timestamps
  created_at        TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at        TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  completed_at      TIMESTAMPTZ
);

-- Indexes
CREATE INDEX idx_ag_sessions_project ON ag_sessions(project_id);
CREATE INDEX idx_ag_sessions_team ON ag_sessions(team_id);
CREATE INDEX idx_ag_sessions_cohort ON ag_sessions(cohort_id);
CREATE INDEX idx_ag_sessions_status ON ag_sessions(status);
CREATE INDEX idx_ag_sessions_created_by ON ag_sessions(created_by);
```

## TABLE 2: ag_insights

Append-only. One row per captured insight. NEVER deleted or merged.

```sql
CREATE TABLE ag_insights (
  id                UUID PRIMARY KEY DEFAULT gen_random_uuid(),

  -- Session reference
  session_id        UUID NOT NULL REFERENCES ag_sessions(id) ON DELETE CASCADE,

  -- Platform references
  project_id        UUID NOT NULL REFERENCES projects(id),
  team_id           UUID NOT NULL REFERENCES teams(id),
  cohort_id         UUID REFERENCES cohorts(id),

  -- Contributor identity (from platform auth — never user-typed)
  user_id           UUID NOT NULL REFERENCES users(id),
  user_role         TEXT NOT NULL CHECK (user_role IN ('founder', 'student', 'mentor', 'admin')),

  -- Tagging
  stakeholder_tag   TEXT NOT NULL,          -- e.g. "Farmer", "Patient", "Retailer"
  dimension         TEXT NOT NULL
                    CHECK (dimension IN ('WHO', 'WHAT', 'WHEN', 'WHERE', 'WHY', 'HOW')),

  -- Content (VERBATIM — never altered)
  raw_text          TEXT NOT NULL,
  language          TEXT NOT NULL DEFAULT 'en'
                    CHECK (language IN ('en', 'hi', 'mixed')),

  -- Pattern signals
  frequency         INT NOT NULL DEFAULT 1,
  signal            TEXT NOT NULL DEFAULT 'LOW'
                    CHECK (signal IN ('HIGH', 'MEDIUM', 'LOW')),
  divergence_flag   BOOLEAN DEFAULT FALSE,   -- TRUE if contradicts another insight

  -- Cluster assignment (set by ClusterAgent)
  cluster_id        UUID REFERENCES ag_clusters(id) ON DELETE SET NULL,

  -- Timestamps
  created_at        TIMESTAMPTZ NOT NULL DEFAULT NOW()

  -- NOTE: No updated_at — this table is append-only
  -- NOTE: No deleted_at — insights are NEVER soft-deleted
);

-- Indexes
CREATE INDEX idx_ag_insights_session ON ag_insights(session_id);
CREATE INDEX idx_ag_insights_user ON ag_insights(user_id);
CREATE INDEX idx_ag_insights_dimension ON ag_insights(dimension);
CREATE INDEX idx_ag_insights_signal ON ag_insights(signal);
CREATE INDEX idx_ag_insights_cluster ON ag_insights(cluster_id);
CREATE INDEX idx_ag_insights_project ON ag_insights(project_id);

-- Append-only enforcement trigger
CREATE OR REPLACE FUNCTION prevent_insight_modification()
RETURNS TRIGGER AS $$
BEGIN
  RAISE EXCEPTION 'ag_insights is append-only. Updates and deletes are not permitted.';
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER enforce_append_only
BEFORE UPDATE OR DELETE ON ag_insights
FOR EACH ROW EXECUTE FUNCTION prevent_insight_modification();
```

## TABLE 3: ag_clusters

Groups of related insights. Generated by ClusterAgent.

```sql
CREATE TABLE ag_clusters (
  id                UUID PRIMARY KEY DEFAULT gen_random_uuid(),

  -- Session reference
  session_id        UUID NOT NULL REFERENCES ag_sessions(id) ON DELETE CASCADE,
  project_id        UUID NOT NULL REFERENCES projects(id),

  -- Cluster metadata
  theme             TEXT NOT NULL,
  primary_dimension TEXT NOT NULL CHECK (primary_dimension IN ('WHO','WHAT','WHEN','WHERE','WHY','HOW')),

  -- Signal data
  signal            TEXT NOT NULL CHECK (signal IN ('HIGH', 'MEDIUM', 'LOW')),
  evidence_count    INT NOT NULL DEFAULT 0,
  contributing_users UUID[] DEFAULT '{}',
  contributing_roles TEXT[] DEFAULT '{}',

  -- Flags
  divergence        BOOLEAN DEFAULT FALSE,
  opportunity_gap   BOOLEAN DEFAULT FALSE,   -- TRUE if cluster contains HOW workarounds

  -- Cluster version (regenerated on each ClusterAgent run)
  cluster_run       INT NOT NULL DEFAULT 1,

  created_at        TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at        TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_ag_clusters_session ON ag_clusters(session_id);
CREATE INDEX idx_ag_clusters_signal ON ag_clusters(signal);
```

## TABLE 4: ag_mind_maps

One synthesised mind map per session (or per synthesis run).

```sql
CREATE TABLE ag_mind_maps (
  id                UUID PRIMARY KEY DEFAULT gen_random_uuid(),

  -- References
  session_id        UUID NOT NULL REFERENCES ag_sessions(id) ON DELETE CASCADE,
  project_id        UUID NOT NULL REFERENCES projects(id),
  team_id           UUID NOT NULL REFERENCES teams(id),
  cohort_id         UUID REFERENCES cohorts(id),
  generated_by      UUID NOT NULL REFERENCES users(id),

  -- Output content
  mind_map_md       TEXT NOT NULL,        -- Full Markdown mind map
  signal_summary    JSONB NOT NULL DEFAULT '[]',
  opportunity_gaps  JSONB NOT NULL DEFAULT '[]',
  divergence_report JSONB DEFAULT NULL,
  coverage_report   JSONB NOT NULL DEFAULT '{}',

  -- Stats at time of generation
  total_insights    INT NOT NULL DEFAULT 0,
  dimensions_covered TEXT[] DEFAULT '{}',
  session_type      TEXT,

  -- Versioning
  version           INT NOT NULL DEFAULT 1,

  created_at        TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_ag_mind_maps_session ON ag_mind_maps(session_id);
CREATE INDEX idx_ag_mind_maps_project ON ag_mind_maps(project_id);
```

## TABLE 5: ag_prompt_versions

Registry of all prompt versions. Enables A/B testing and rollback.

```sql
CREATE TABLE ag_prompt_versions (
  id                UUID PRIMARY KEY DEFAULT gen_random_uuid(),

  version_tag       TEXT NOT NULL UNIQUE,   -- e.g. 'v2', 'v2.1', 'v3-hindi-test'
  agent_name        TEXT NOT NULL
                    CHECK (agent_name IN ('intake', 'capture', 'cluster', 'synthesis', 'orchestrator')),
  prompt_text       TEXT NOT NULL,
  changelog         TEXT,
  is_active         BOOLEAN NOT NULL DEFAULT FALSE,
  is_default        BOOLEAN NOT NULL DEFAULT FALSE,

  created_by        UUID REFERENCES users(id),
  created_at        TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  deprecated_at     TIMESTAMPTZ
);

-- Only one default per agent at a time
CREATE UNIQUE INDEX idx_one_default_per_agent
ON ag_prompt_versions(agent_name)
WHERE is_default = TRUE;

CREATE INDEX idx_ag_prompt_version_tag ON ag_prompt_versions(version_tag);
CREATE INDEX idx_ag_prompt_active ON ag_prompt_versions(is_active);
```

## ROW LEVEL SECURITY (RLS)

All tables use Supabase RLS, aligned to INCUBX platform roles.

```sql
-- Enable RLS on all Antigravity tables
ALTER TABLE ag_sessions ENABLE ROW LEVEL SECURITY;
ALTER TABLE ag_insights ENABLE ROW LEVEL SECURITY;
ALTER TABLE ag_clusters ENABLE ROW LEVEL SECURITY;
ALTER TABLE ag_mind_maps ENABLE ROW LEVEL SECURITY;

-- ag_sessions: founders/students see own team only, mentors see assigned cohort
CREATE POLICY "team_member_access" ON ag_sessions
  FOR SELECT USING (
    team_id IN (
      SELECT team_id FROM team_members WHERE user_id = auth.uid()
    )
  );

CREATE POLICY "mentor_cohort_access" ON ag_sessions
  FOR SELECT USING (
    cohort_id IN (
      SELECT cohort_id FROM cohort_mentors WHERE mentor_id = auth.uid()
    )
  );

CREATE POLICY "admin_full_access" ON ag_sessions
  FOR ALL USING (
    EXISTS (SELECT 1 FROM users WHERE id = auth.uid() AND role = 'admin')
  );

-- ag_insights: same access pattern, plus INSERT allowed for active team members
CREATE POLICY "team_insert_insights" ON ag_insights
  FOR INSERT WITH CHECK (
    session_id IN (
      SELECT id FROM ag_sessions
      WHERE team_id IN (
        SELECT team_id FROM team_members WHERE user_id = auth.uid()
      )
    )
    AND user_id = auth.uid()
  );
```

## MIGRATION ORDER

Run in this sequence to avoid FK violations:

```
1. ag_prompt_versions    (no dependencies)
2. ag_sessions           (depends on: projects, teams, cohorts, users)
3. ag_clusters           (depends on: ag_sessions, projects)
4. ag_insights           (depends on: ag_sessions, ag_clusters, projects, teams, cohorts, users)
5. ag_mind_maps          (depends on: ag_sessions, projects, teams, cohorts, users)
```

## SCHEMA EXTENSION NOTE FOR INCUBX DEVS

This schema assumes the following columns exist in your current tables.
If they don't, they need to be added before migration:

```sql
-- Required on existing `users` table
-- id UUID, role TEXT

-- Required on existing `teams` table
-- id UUID, cohort_id UUID

-- Required on existing `projects` table
-- id UUID, team_id UUID

-- Required on existing `cohorts` table
-- id UUID
```

If column names differ in your schema, update the FK references accordingly.
